# Cases for turn-queue, with outcomes written by hand from the rules in README.md.
# prepare.py turns the case starts and actions into the program's script.
# expected.py compares each record the program writes with the outcome here.
#
# case <name>                    starts a case
# start <world>                  the world before the first step
# <action> => <status> <resolved> <world>
#   action:   send <player> <source> <destination> <ships> <delay>, begin or end
#   status:   accepted or rejected
#   resolved: slots that landed in this step, such as 0 or 0,1, or -
#   world:    <turn>/<limit> <phase> <ships on planet 0>,<planet 1> <slot 0> <slot 1>
#   phase:    complete, commands or finished
#   slot:     - when free, else <destination>:<ships>@<arrival turn>

case queue-lifecycle
start                => 1/4 commands 20,5 - -
send 1 0 1 3 1       => accepted - 1/4 commands 17,5 1:3@2 -
send 1 0 1 4 2       => accepted - 1/4 commands 13,5 1:3@2 1:4@3
send 1 0 1 3 1       => rejected - 1/4 commands 13,5 1:3@2 1:4@3
end                  => accepted - 1/4 complete 13,5 1:3@2 1:4@3
begin                => accepted 0 2/4 commands 13,8 - 1:4@3
send 1 1 0 2 1       => accepted - 2/4 commands 13,6 0:2@3 1:4@3
end                  => accepted - 2/4 complete 13,6 0:2@3 1:4@3
begin                => accepted 0,1 3/4 commands 15,10 - -
end                  => accepted - 3/4 complete 15,10 - -
begin                => accepted - 4/4 commands 15,10 - -
end                  => accepted - 4/4 finished 15,10 - -
begin                => rejected - 4/4 finished 15,10 - -

case overdue-saturation-last-slot
start                => 2/4 complete 250,5 - 0:10@1
begin                => accepted 1 3/4 commands 255,5 - -

case maximum-turn
start                => 65534/65535 commands 20,5 - -
send 1 0 1 3 1       => accepted - 65534/65535 commands 17,5 1:3@65535 -
end                  => accepted - 65534/65535 complete 17,5 1:3@65535 -
begin                => accepted 0 65535/65535 commands 17,8 - -
send 1 0 1 3 1       => rejected - 65535/65535 commands 17,8 - -
end                  => accepted - 65535/65535 finished 17,8 - -
begin                => rejected - 65535/65535 finished 17,8 - -

case send-all
start                => 1/4 commands 20,5 - -
send 1 0 1 20 3      => accepted - 1/4 commands 0,5 1:20@4 -

# Each rejected send leaves the world unchanged. A valid send follows it.
case wrong-player

send 1 0 1 3 1       => accepted - 1/4 commands 17,5 1:3@2 -

case same-planet

# An operation in the wrong phase is rejected and changes nothing.
